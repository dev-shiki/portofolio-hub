Case Study 01: Universal Workflow Engine

Application-layer AES-256-GCM PII encryption + HMAC-SHA256 blind indexing + append-only audit log. 

// User: PII encrypted at application layer before DB write
model User {
  emailCiphertext  String  @map("email")       // 🔒 AES-256-GCM
  emailBlindIndex  String  @unique @map("email_bidx") // 🔎 HMAC-SHA256
}

// Generic BPE request — any approval domain
model WorkflowRequest {
  domain        String         // "procurement" | "hr" | "maintenance"
  payload       Json           // typed domain-specific fields
  currentStatus WorkflowStatus @default(draft)

  @@index([domain])
  @@index([currentStatus])
}

// Append-only step log — INSERT only, never UPDATE or DELETE
model ApprovalHistory {
  previousStatus WorkflowStatus
  newStatus      WorkflowStatus
  executedAt     DateTime @default(now())

  @@index([requestId])
}

Case Study 02: Immutable Ledger Trigger

PL/pgSQL trigger enforcing append-only semantics at PostgreSQL engine level. 

CREATE OR REPLACE FUNCTION enforce_ledger_immutability()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'UPDATE' THEN
        RAISE EXCEPTION
            'SECURITY VIOLATION: Ledger entry % cannot be modified. Attempted by session user "%".',
            OLD.id, session_user
            USING ERRCODE = 'restrict_violation';  -- SQLSTATE 23001

    ELSIF TG_OP = 'DELETE' THEN
        RAISE EXCEPTION
            'SECURITY VIOLATION: Ledger entry % cannot be deleted. Attempted by session user "%".',
            OLD.id, session_user
            USING ERRCODE = 'restrict_violation';
    END IF;

    RETURN NULL;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER trg_ledger_immutability
    BEFORE UPDATE OR DELETE ON event_wallet_ledger
    FOR EACH ROW EXECUTE FUNCTION enforce_ledger_immutability();
